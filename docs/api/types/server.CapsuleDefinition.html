    Type Alias CapsuleDefinition<Schema>

    Top-level Capsule definition passed to capsule().

    The Capsule is the deployable unit: schema, server handlers, middleware, and hooks bundled with the client, config, and runtime data boundary.

    type CapsuleDefinition<Schema extends SchemaDefinition = SchemaDefinition> = {
        accessKeys?: { scopes: readonly string[] };
        auth?: {
            reauthentication?: {
                authorize?: (
                    ctx: { auth: AuthContext; db: ReadOnlyDatabaseFromSchema<Schema> },
                    purpose: string,
                ) => MaybePromise<boolean>;
                purposes: Record<string, { maxAgeSeconds: number }>;
            };
            registration?: RegistrationAdmission<Schema>;
        };
        emailEvents?: EmailEventDefinition<EmailEventHandler<Schema>>;
        endpoints?: Record<
            string,
            EndpointDefinition<
                EndpointHandler<Schema>
                | AuthGuardedHandler<(...args: any[]) => any>,
            >,
        >;
        files?: CapsuleFilesDefinition<Schema>;
        hooks?: CapsuleHooks<Schema>;
        jobs?: Record<string, JobDefinition<any>>;
        journey?: {
            capture?: {
                focus?: boolean;
                interactions?: boolean;
                navigation?: boolean;
            };
            enabled: true;
            ttlSeconds?: number;
        };
        messages?: Record<
            string,
            MessageDefinition<
                MessageHandler<Schema>
                | AuthGuardedHandler<(...args: any[]) => any>,
            >,
        >;
        middleware?: ContextMiddleware<Schema>[];
        mutations?: Record<
            string,
            MutationDefinition<
                | MutationHandler<Schema, any[]>
                | AuthGuardedHandler<(...args: any[]) => any>,
            >,
        >;
        name: string;
        queries?: Record<
            string,
            QueryDefinition<
                | QueryHandler<Schema, any>
                | AuthGuardedHandler<(...args: any[]) => any>,
            >,
        >;
        schedules?: Record<string, ScheduleDefinition>;
        schema?: Schema;
        stripeEvents?:
            | StripeEventDefinition<StripeEventHandler<Schema>>
            | AtomicStripeEventDefinition<AtomicStripeEventHandler<Schema>>;
        teamBilling?: TeamBillingDefinition<Schema>;
        teams?: {
            appRoles?: readonly string[];
            admitJoin?(
                ctx: TeamJoinAdmissionContext<Schema>,
                input: TeamJoinAdmissionInput,
            ): MaybePromise<TeamJoinAdmissionDecision>;
        };
    }
